demolh 127.0.0.1

openssl genrsa -out ca.key 2048
openssl req -x509 -new -nodes -key ca.key -sha256 -days 3650 -out ca.crt -subj "/CN=MyCloudCA"

openssl genrsa -out server.key 2048
openssl req -new -key server.key -out server.csr -subj "/CN=demolh.hsecontest.ru/O=CMTLab" -addext "subjectAltName=DNS:demolh.hsecontest.ru"
openssl x509 -req -in server.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out server.crt -days 365 -sha256 -copy_extensions copy

openssl genrsa -out client.key 2048
openssl req -new -key client.key -out client.csr -subj "/CN=admin-client-cert/O=CMTLab"
openssl x509 -req -days 365 -in client.csr -CA ca.crt -CAkey ca.key -CAcreateserial -out client.crt -extfile <(echo "extendedKeyUsage=clientAuth")
openssl pkcs12 -export -out client.p12 -inkey client.key -in client.crt -certfile ca.crt


ssl_client_certificate /etc/nginx/certs/ca.crt;
ssl_verify_client on;
ssl_verify_depth 1;
